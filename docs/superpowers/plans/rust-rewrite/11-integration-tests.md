## Task 11.2: Backward Compatibility Tests

Each test verifies that Rust code can read Python-written artifacts.

- [ ] **Write compatibility tests**

```rust
// tests/compat/read_python_fixtures.rs

#[test]
fn read_python_lockfile() {
    let json = include_str!("fixtures/lockfile_python.json");
    let lockfile = ap_fetcher::lockfile::LockfileManager::parse(json).unwrap();
    assert!(lockfile.agents.contains_key("code-reviewer"));
    let entry = &lockfile.agents["code-reviewer"];
    assert_eq!(entry.version, "1.2.0");
}

#[test]
fn read_python_config() {
    let toml = include_str!("fixtures/config_python.toml");
    let config = ap_core::config::loader::ConfigLoader::load_from_str(toml).unwrap();
    assert_eq!(config.models.default, "openai:gpt-4o");
    assert!(config.models.providers.contains_key("deepseek"));
}

#[test]
fn read_python_sources() {
    let yaml = include_str!("fixtures/sources_python.yaml");
    let sources = ap_fetcher::sources::SourceManager::parse(yaml).unwrap();
    assert!(!sources.is_empty());
}

#[test]
fn read_python_manifest() {
    let yaml = include_str!("fixtures/manifest_python.yaml");
    let manifest: ap_core::models::agent::AgentManifest = serde_yaml::from_str(yaml).unwrap();
    assert!(!manifest.name.is_empty());
}

#[test]
fn read_python_evolution_db() {
    // Copy fixture DB to temp location, open with Rust store
    let db_path = std::path::Path::new("tests/compat/fixtures/evolution_python.db");
    if db_path.exists() {
        let store = ap_evolution::store::EvolutionStore::open(db_path).unwrap();
        let skills = store.list_skills().unwrap();
        assert!(!skills.is_empty());
    }
}

#[test]
fn read_python_dsl_toml() {
    let toml = include_str!("fixtures/dsl_python.toml");
    let dag = ap_core::orchestration::dsl::OrchestrationDsl::parse(toml).unwrap();
    assert!(!dag.tasks.is_empty());
}
```
